#15. triangle:

#20. sum squares:
def sos(n):
    s = 0
    for d in str(n):
        s += int(d) ** 2
    return s
print(sos(1))
print(sos(9))
print(sos(11))
print(sos(121))

#21. Fibonacci:
a, b = 0, 1
fib = []
while a <= 50:
    fib.append(a)
    a, b = b, a + b
fib = [x for x in fib if x != 0 and x <= 34]
print(" ".join(map(str, fib)))

#22. primes:
for num in range(2, 101):
    prime = True
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            prime = False
            break
    if prime:
        print(num, end=" ")
print()

#28. reverse:
num = int(input("num: "))
rev = 0
temp = abs(num)
while temp:
    rev = rev * 10 + temp % 10
    temp //= 10
if num < 0:
    rev = -rev
print(rev)
