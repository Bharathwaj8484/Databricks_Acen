#2. count chars:
s = input("str: ")
lc = sum(1 for ch in s if ch.islower())
uc = sum(1 for ch in s if ch.isupper())
dig = sum(1 for ch in s if ch.isdigit())
sym = len(s) - lc - uc - dig
print("Chars =", lc + uc)
print("Digits =", dig)
print("Symbol =", sym)

#3. sum & avg:
import re
s = input("str: ")
nums = re.findall(r'\d+', s)
nums = [int(x) for x in nums]
if nums:
    print("sum is", sum(nums))
    print("average is", sum(nums)/len(nums))
else:
    print("sum is 0")
    print("average is 0")

#5. max word:
s = input("str: ")
words = s.split()
words = [w.strip(".,") for w in words]
max_word = max(set(words), key=words.count)
print(max_word, "occurring", words.count(max_word), "times")

#14. phone book:
phone = {}
while True:
    cmd = input("cmd (add/find/del/exit): ")
    if cmd == "add":
        name = input("name: ")
        num = input("number: ")
        phone[name] = num
    elif cmd == "find":
        name = input("name: ")
        print(phone.get(name, "not found"))
    elif cmd == "del":
        name = input("name: ")
        if name in phone:
            del phone[name]
            print("deleted")
        else:
            print("not found")
    elif cmd == "exit":
        break
    else:
        print("invalid")
