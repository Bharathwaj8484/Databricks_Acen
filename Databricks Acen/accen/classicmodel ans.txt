-- 1. Retrieve the total number of Customers.
SELECT COUNT(*) AS total_customers
FROM customers;

-- 2. Retrieve total number of orders.
SELECT COUNT(*) AS total_orders
FROM orders;

-- 3. Retrieve total number of orders per customer.
SELECT customerNumber, COUNT(*) AS total_orders
FROM orders
GROUP BY customerNumber;

-- 4. Retrieve total number of orders per status.
SELECT status, COUNT(*) AS total_orders
FROM orders
GROUP BY status;

-- 5. Retrieve total number of orders per order date.
SELECT orderDate, COUNT(*) AS total_orders
FROM orders
GROUP BY orderDate;

-- 6. Retrieve the earliest order date.
SELECT MIN(orderDate) AS earliest_order_date
FROM orders;

-- 7. Retrieve latest order date.
SELECT MAX(orderDate) AS latest_order_date
FROM orders;

-- 8. Retrieve total number of orders per year.
SELECT YEAR(orderDate) AS year, COUNT(*) AS total_orders
FROM orders
GROUP BY YEAR(orderDate);

-- 9. Retrieve total number of orders per month.
SELECT DATE_FORMAT(orderDate, '%Y-%m') AS month, COUNT(*) AS total_orders
FROM orders
GROUP BY DATE_FORMAT(orderDate, '%Y-%m');

-- 10. Retrieve total number of orders per day of the week.
SELECT DAYNAME(orderDate) AS day_of_week, COUNT(*) AS total_orders
FROM orders
GROUP BY DAYNAME(orderDate);

-- 11. Retrieve total number of orders per quarter.
SELECT QUARTER(orderDate) AS quarter, COUNT(*) AS total_orders
FROM orders
GROUP BY QUARTER(orderDate);

-- 12. Retrieve total number of orders per customer and status. Sort by Customer number.
SELECT customerNumber, status, COUNT(*) AS total_orders
FROM orders
GROUP BY customerNumber, status
ORDER BY customerNumber;

-- 13. Retrieve total number of orders per customer per year. Sort by customer number.
SELECT customerNumber, YEAR(orderDate) AS year, COUNT(*) AS total_orders
FROM orders
GROUP BY customerNumber, YEAR(orderDate)
ORDER BY customerNumber;

-- 14. Retrieve the total number of orders for the first 10 even weeks.
SELECT WEEK(orderDate, 1) AS week_number, COUNT(*) AS total_orders
FROM orders
WHERE WEEK(orderDate, 1) % 2 = 0
GROUP BY WEEK(orderDate, 1)
ORDER BY week_number
LIMIT 10;

-- 15. Retrieve total number of orders per status per year. Sort by status.
SELECT status, YEAR(orderDate) AS year, COUNT(*) AS total_orders
FROM orders
GROUP BY status, YEAR(orderDate)
ORDER BY status;

-- 16. Retrieve total number of orders per customer per year and quarter for the years 2004 and 2005.
SELECT customerNumber, YEAR(orderDate) AS year, QUARTER(orderDate) AS quarter, COUNT(*) AS total_orders
FROM orders
WHERE YEAR(orderDate) IN (2004, 2005)
GROUP BY customerNumber, YEAR(orderDate), QUARTER(orderDate);

-- 17. Retrieve the top 5 longest customer names along with their lengths.
SELECT customerName, LENGTH(customerName) AS name_length
FROM customers
ORDER BY name_length DESC
LIMIT 5;

-- 18. Extract the first 5 characters of customer names and convert to uppercase.
SELECT UPPER(LEFT(customerName, 5)) AS customer_name_prefix
FROM customers;

-- 19. Replace 'Ltd' with 'Limited' in customer names and display only those modified customer names.
SELECT REPLACE(customerName, 'Ltd', 'Limited') AS modified_customer_name
FROM customers
WHERE customerName LIKE '%Ltd%';

-- 20. Round the credit limit to the nearest thousand and display the customer name along with the rounded credit limit.
SELECT customerName, ROUND(creditLimit, -3) AS rounded_credit_limit
FROM customers;

-- 36. Retrieve the customer numbers, names, and total sales amounts for customers with sales above $10,000.
SELECT
    c.customerNumber,
    c.customerName,
    SUM(od.quantityOrdered * od.priceEach) AS totalSalesAmount
FROM
    customers c
    JOIN orders o ON c.customerNumber = o.customerNumber
    JOIN orderdetails od ON o.orderNumber = od.orderNumber
GROUP BY
    c.customerNumber,
    c.customerName
HAVING
    totalSalesAmount > 10000;

-- 37. Retrieve the customer numbers, names, and average sales values for customers with an average order value exceeding $3,500.

FROM
    customers c
    JOIN orders o ON c.customerNumber = o.customerNumber
    JOIN orderdetails od ON o.orderNumber = od.orderNumber
GROUP BY
    c.customerNumber,
    c.customerName
HAVING
    averageOrderValue > 3500;

-- 39. Retrieve the product codes, names, and total sales for products with sales above $180,000.

    p.productCode,

    SUM(od.quantityOrdered * od.priceEach) AS totalSales
FROM
    products p
    JOIN orderdetails od ON p.productCode = od.productCode
GROUP BY
    p.productCode,
    p.productName
HAVING
    totalSales > 180000;

-- 40. Retrieve the product codes, names, and average quantities ordered for products with an average above 30.
SELECT
    p.productCode,
    p.productName,
    AVG(od.quantityOrdered) AS averageQuantityOrdered
FROM
    products p
    JOIN orderdetails od ON p.productCode = od.productCode
GROUP BY
    p.productCode,
    p.productName
HAVING
    AVG(od.quantityOrdered) > 30;

-- 41. Retrieve the employee numbers, first and last names, and sales amounts for employees with sales above $500,000.

    e.lastName

    salesAmount > 500000;

-- 46. Retrieve the office name and sales amount for offices (cities) with total sales exceeding $300,000.
SELECT
    o.city AS officeName,
    SUM(od.quantityOrdered * od.priceEach) AS salesAmount
FROM
    offices o
    JOIN employees e ON o.officeCode = e.officeCode
    JOIN customers c ON e.employeeNumber = c.salesRepEmployeeNumber
    JOIN orders odr ON c.customerNumber = odr.customerNumber
    JOIN orderdetails od ON odr.orderNumber = od.orderNumber
GROUP BY
    o.city
HAVING
    salesAmount > 300000;

-- 70. Retrieve the names of customers with credit limits higher than the average credit limit in their country.
SELECT
    c.customerName
FROM
    customers c
WHERE
    c.creditLimit > (
        SELECT AVG(c2.creditLimit)
        FROM customers c2
        WHERE c2.country = c.country
    );
